rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null
        && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    // Keys the user's own browser is allowed to write on their profile.
    function userOwnedKeys() {
      return [
        "cashBalance", "airtimeBalance", "totalEarned", "todayEarnings",
        "name", "referralCode", "photoURL", "telegramJoined",
        "submissions", "ledger", "withdrawalList",
        "creditedReferrals", "appliedCredits",
        "rev", "lastSyncAt"
      ];
    }

    match /admins/{uid} {
      allow read: if request.auth != null && (request.auth.uid == uid || isAdmin());
      allow write: if isAdmin();
    }

    // Tasks + settings, published as ONE document by the admin panel.
    // Must be readable without sign-in: the app loads it before auth.
    match /config/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /settings/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /adminLogs/{logId} {
      allow read: if isAdmin();
      allow create: if isAdmin();
      allow update, delete: if false;
    }

    match /users/{uid} {
      allow read: if request.auth != null && (request.auth.uid == uid || isAdmin());

      allow create: if request.auth != null && request.auth.uid == uid
        && request.resource.data.airtimeBalance == 0
        && request.resource.data.totalEarned == request.resource.data.cashBalance
        && request.resource.data.todayEarnings == request.resource.data.cashBalance
        && (
          !exists(/databases/$(database)/documents/settings/site)
            ? request.resource.data.cashBalance == 0
            : request.resource.data.cashBalance ==
                get(/databases/$(database)/documents/settings/site).data.get('welcomeBonus', 0)
        );

      allow update: if request.auth != null && (
        // The user's own device syncing its profile back.
        (request.auth.uid == uid
          && request.resource.data.diff(resource.data).affectedKeys()
              .hasOnly(userOwnedKeys()))
        ||
        // Admin adjustments. pendingCredits is admin-only and is the
        // channel refunds travel through, because the user's browser
        // holds the newer copy of their balance.
        (isAdmin()
          && request.resource.data.diff(resource.data).affectedKeys()
              .hasOnly(["cashBalance", "airtimeBalance", "totalEarned",
                        "todayEarnings", "banned", "pendingCredits"]))
      );

      allow delete: if isAdmin();
    }

    match /tasks/{taskId} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }

    // Legacy. The app no longer writes these; kept readable so old
    // records stay visible in the admin panel.
    match /taskSubmissions/{subId} {
      allow read: if request.auth != null && (resource.data.userId == request.auth.uid || isAdmin());
      allow write: if isAdmin();
    }

    match /transactions/{txId} {
      allow read: if request.auth != null && (resource.data.userId == request.auth.uid || isAdmin());
      allow write: if isAdmin();
    }

    match /referralCodes/{code} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.uid == request.auth.uid;
      allow update, delete: if false;
    }

    match /referrals/{refId} {
      allow read: if request.auth != null
        && (resource.data.referredUserId == request.auth.uid
            || resource.data.referrerId == request.auth.uid
            || isAdmin());

      allow create: if request.auth != null
        && request.resource.data.referredUserId == request.auth.uid
        && request.resource.data.referrerId != request.auth.uid
        && request.resource.data.rewardCredited == false
        && request.resource.data.status == "pending";

      // Users never write referral documents any more — whether a
      // referral was paid is recorded on the referrer's own profile.
      allow update: if isAdmin();
      allow delete: if false;
    }

    match /withdrawals/{wId} {
      allow read: if request.auth != null && (resource.data.userId == request.auth.uid || isAdmin());

      allow create: if request.auth != null
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.status == "pending"
        && request.resource.data.refunded == false;

      allow update: if isAdmin()
        && request.resource.data.diff(resource.data).affectedKeys()
            .hasOnly(["status", "refunded", "processedAt", "processedBy",
                      "providerReference", "processingAt", "processingBy"]);

      allow delete: if false;
    }
  }
}
